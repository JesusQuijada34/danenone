cmake_minimum_required(VERSION 3.16)
project(influent_danenone_qt_shell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Svg)
add_executable(influent-danenone-qt-shell main.cpp)
target_link_libraries(influent-danenone-qt-shell PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg)
install(TARGETS influent-danenone-qt-shell RUNTIME DESTINATION bin)
